% Encoding & language

% ---------------------------------------------------
% Page layout & margins

\usepackage{geometry}
\geometry{a4paper, margin=2.5cm}

% ---------------------------------------------------
% Chapter titles & headers/footers
% ---------------------------------------------------
\usepackage{titlesec}        % Personnalisation des titres
\titleformat{\chapter}[display]  % label sur sa propre ligne, titre en dessous
  {\normalfont\huge\bfseries}    % Style du texte
  {\chaptername\ \thechapter}    % "Chapitre 1"
  {0pt}                          % Aucun espace entre le label et le titre
  {\Huge}                        % Style du titre

% Number headings down to subsubsection level, so that Chapter 4's
% subsubsections appear as 4.1.2.1, 4.1.2.2, ... rather than unnumbered.
\setcounter{secnumdepth}{3}

\usepackage{fancyhdr}
\pagestyle{fancy}
% fancyhdr's own defaults include \fancyfoot[C]{\thepage}. They must be cleared,
% or the folio is printed twice on every body page: once top-right by the
% \fancyhead[R] below and once again at the foot.
\fancyhf{}
% The 12pt class leaves \headheight at 12pt, but the running head needs 14.5pt;
% without this fancyhdr warns on every \chapter. Shrink \topmargin by the same
% amount so the text block does not shift down the page. NOTE: if a chapter
% title ever grows long enough to wrap the running head onto a second line,
% fancyhdr will ask for ~27.5pt here (and -15.5pt below) -- or shorten the head
% with \chapter[short title]{long title}, which is usually the better answer.
\setlength{\headheight}{14.5pt}
\addtolength{\topmargin}{-2.5pt}
% Seuls les chapitres numerotes portent "Chapter N -- " dans l'en-tete.
% Les chapitres non numerotes (Contents, Introduction, Conclusion...) n'affichent
% que leur titre, car \chapter* n'appelle pas \chaptermark.
\renewcommand{\chaptermark}[1]{\markboth{\chaptername\ \thechapter\ --\ #1}{}}

\fancyhead[R]{\thepage}
\fancyhead[L]{\leftmark}
% Chapter-opening pages use \pagestyle{plain}; keep their folio in the same
% place as everywhere else instead of the class default (bottom centre).
\fancypagestyle{plain}{%
  \fancyhf{}%
  \fancyhead[R]{\thepage}%
  \renewcommand{\headrulewidth}{0pt}%
}

% ---------------------------------------------------

% author's draft. Numbering on [section] would instead give 5.3.1, 5.3.2.
\newtheorem{proposition}{Proposition}[chapter]
\newtheorem{corollary}[proposition]{Corollary}

% ---------------------------------------------------
% References, URLs, acronyms, multiple bibliographies
% ---------------------------------------------------
% xurl extends url.sty so a URL may break after ANY character. Plain url.sty
% breaks only at a handful of punctuation marks, which left the long Concordia
% URLs unable to fill their line: the text line in front of each one came out
% underfull and visibly gappy ("https : / /").
\usepackage{xurl}
\usepackage[footnote]{acronym}
\usepackage[resetlabels,labeled]{multibib}
% The suffix given to \newcites is what the "labeled" option prints in the running
% text. It is "N" so that web sources cite as [N1], [N2], ... -- a clean second
% numeric series alongside the main bibliography's [1], [2], ... -- rather than
% the ad-hoc [URL1] form the previous suffix produced. The commands defined are
% therefore \citeN, \bibliographyN and \bibliographystyleN.
\newcites{N}{Netography} % single clean "Netography" bibliography

% ---------------------------------------------------
% Arabic -- required by the EPT cover page (pg-EPT.tex), which sets the
% ministry / university header in Arabic beside its French counterpart.
% Compile with pdflatex. arabtex must come AFTER inputenc/fontenc and its
% companion utf8.sty is what lets \setcode{utf8} accept literal Arabic script.
% auxhook is pulled in later by hyperref, and it patches \document. arabtex
% replaces \document, so an auxhook arriving afterwards finds a definition it
% does not recognise and falls back with "Cannot patch \document". Loading it
% here, while \document is still pristine, lets the patch take; arabtex then
% captures the already-patched version in its own \a@document.
\usepackage{auxhook}
\usepackage{arabtex}
\usepackage{utf8}
% The cover's Arabic blocks are quarantined in their own file, so that
% pg-EPT.tex stays free of arabtex-sensitive syntax and can be edited freely.
\input{chapters/cover-arabic}

% Subfiles for modular document
% ---------------------------------------------------
\usepackage{subfiles}

% ---------------------------------------------------
% PDF inclusion
% ---------------------------------------------------
\usepackage{pdfpages}

% ---------------------------------------------------

% ---------------------------------------------------
\usepackage[hidelinks]{hyperref}

% Last-resort stretch for lines TeX cannot otherwise break. LaTeX will not
% hyphenate inside an already-hyphenated compound, so words such as
% "retrieval-augmented" would otherwise overhang the right margin.
\setlength{\emergencystretch}{1em}

% ---------------------------------------------------
% Lengths for figures (kept from your original)

\newlength\figureheight
